import requests
import pytest


# =========================
# Base API-класс
# =========================
class BaseAPI:
    BASE_URL = "https://practice-api-qa.herokuapp.com"
    product_id = None  # сюда можно записать id пользователя извне

    def get(self, endpoint, **kwargs):
        return requests.get(f"{self.BASE_URL}{endpoint}", **kwargs)

    def post(self, endpoint, data=None, json=None, **kwargs):
        return requests.post(f"{self.BASE_URL}{endpoint}", data=data, json=json, **kwargs)

    def put(self, endpoint, data=None, json=None, **kwargs):
        return requests.put(f"{self.BASE_URL}{endpoint}", data=data, json=json, **kwargs)

    def delete(self, endpoint, **kwargs):
        return requests.delete(f"{self.BASE_URL}{endpoint}", **kwargs)


# =========================
# API для пользователей
# =========================
class ProductAPI(BaseAPI):

    def create_product(self, name, description, price, quantity):
        return self.post("/api/products",
                         json={"name": name, "description": description, "price": price, "quantity": quantity})

    def get_product(self, product_id):
        return self.get(f"/api/products/{product_id}")

    def delete_product(self, product_id):
        return self.delete(f"/api/products/{product_id}")

    def remember_product(self, product_id):
        self.product_id = product_id


# =========================
# Фикстуры
# =========================
@pytest.fixture
def api():
    """Возвращает объект ProductsAPI"""
    return ProductAPI()


@pytest.fixture
def new_product(api):
    """Setup-only: создаем пользователя без удаления"""
    response = api.create_product("rulon", "oboev", 1, 1)
    product_id = response.json()["data"]["id"]
    api.remember_product(product_id)
    return product_id


@pytest.fixture
def delete_product(api: ProductAPI):
    """Удаляет пользователя по api.product_id, product_id задается отдельно в тесте"""
    yield
    print(api.product_id)
    if api.product_id is not None:
        api.delete_product(api.product_id)


@pytest.fixture
def setup_teardown_product(new_product, delete_product):
    yield new_product


# =========================
# Тесты CRUD
# =========================
# @pytest.mark.parametrize("productname,password", [("product1", "pass1"), ("product2", "pass2")])
def test_create_product(api, delete_product):
    """Тест создания пользователя + удаление через delete_product"""
    response = api.create_product("rulon1", "oboev", 1, 1)
    assert response.status_code == 201
    product_id = response.json()["data"]["id"]
    api.remember_product(product_id)


def test_get_product(api, setup_teardown_product):
    product_id = setup_teardown_product
    response = api.get_product(product_id)
    assert response.status_code == 200
    assert response.json()["data"]["id"] == setup_teardown_product
    print(f"======== {response.json()['data']['id']} =======")


def test_delete_product(api, new_product):
    del_resp = api.delete_product(new_product)
    assert del_resp.status_code == 200
    assert api.get_product(new_product).status_code == 404
